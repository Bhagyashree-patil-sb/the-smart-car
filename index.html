About The Project

The Smart Navigation Robot is built using ESP32, L298N motor driver, BO motors, ultrasonic sensor, and servo motor. It supports Bluetooth control through a mobile application.

It can operate in manual mode, auto obstacle avoidance mode, and predefined route navigation modes like Home, School, Office, and Hospital.

Working of the Project

1. User sends command via mobile app.
2. ESP32 receives Bluetooth signal.
3. Motor driver controls wheels.
4. Ultrasonic sensor detects obstacles.
5. Servo scans left and right.
6. System selects safe path.
7. Robot moves accordingly.

Arduino / ESP32 Source Code


#include <BluetoothSerial.h>
#include <ESP32Servo.h>

BluetoothSerial SerialBT;
Servo scanner;

#define ENA 14
#define IN1 27
#define IN2 26

#define ENB 12
#define IN3 25
#define IN4 33

#define TRIG 5
#define ECHO 18

#define SERVO_PIN 13

#define CH_A 0
#define CH_B 1

char author[] = {83,117,114,101,115,104,32,83,104,101,114,105,107,97,114,0};

int motorSpeed = 180;

bool autoMode = false;
bool routeMode = false;
bool routeRunning = false;
int routeID = 0;

unsigned long lastCmdTime = 0;

void setSpeed(int s) {
  motorSpeed = constrain(s, 0, 255);
  ledcWrite(CH_A, motorSpeed);
  ledcWrite(CH_B, motorSpeed);
}

void stopCar() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void forward() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void backward() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void left() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void right() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

long getDistance() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH, 30000);
  if (duration == 0) return 999;

  return duration * 0.034 / 2;
}

void autoDrive() {
  scanner.write(90);
  delay(200);

  long front = getDistance();

  if (front > 25) {
    forward();
    return;
  }

  stopCar();
  delay(200);

  scanner.write(150);
  delay(300);
  long leftD = getDistance();

  scanner.write(30);
  delay(300);
  long rightD = getDistance();

  scanner.write(90);

  if (leftD > rightD) {
    left();
  } else {
    right();
  }

  delay(400);
  stopCar();
}

void routeHome() {
  forward(); delay(5000); stopCar();
  right(); delay(700); stopCar();
  forward(); delay(3000); stopCar();
}

void routeSchool() {
  forward(); delay(3000); stopCar();
  left(); delay(600); stopCar();
  forward(); delay(2000); stopCar();
}

void routeOffice() {
  right(); delay(800); stopCar();
  forward(); delay(2500); stopCar();
  left(); delay(500); stopCar();
  forward(); delay(2000); stopCar();
}

void routeHospital() {
  forward(); delay(2000); stopCar();
  left(); delay(700); stopCar();
  forward(); delay(4000); stopCar();
  right(); delay(600); stopCar();
  forward(); delay(1000); stopCar();
}

void setup() {
  Serial.begin(115200);
  SerialBT.begin("ESP32_CAR");

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  scanner.attach(SERVO_PIN);
  scanner.write(90);

  ledcSetup(CH_A, 5000, 8);
  ledcAttachPin(ENA, CH_A);

  ledcSetup(CH_B, 5000, 8);
  ledcAttachPin(ENB, CH_B);

  setSpeed(motorSpeed);
}

void loop() {
  if (SerialBT.available()) {
    char cmd = SerialBT.read();

    autoMode = false;
    routeMode = false;
    routeRunning = false;

    if (cmd == 'A') autoMode = true;
    if (cmd == '1') { routeMode = true; routeID = 1; }
    if (cmd == '2') { routeMode = true; routeID = 2; }
    if (cmd == '3') { routeMode = true; routeID = 3; }
    if (cmd == '4') { routeMode = true; routeID = 4; }

    if (!autoMode && !routeMode) {
      if (cmd == 'F') forward();
      if (cmd == 'B') backward();
      if (cmd == 'L') left();
      if (cmd == 'R') right();
      if (cmd == 'S') stopCar();
    }
  }

  if (autoMode) autoDrive();

  if (routeMode && !routeRunning) {
    routeRunning = true;

    if (routeID == 1) routeHome();
    if (routeID == 2) routeSchool();
    if (routeID == 3) routeOffice();
    if (routeID == 4) routeHospital();

    routeRunning = false;
    routeMode = false;
  }
}
